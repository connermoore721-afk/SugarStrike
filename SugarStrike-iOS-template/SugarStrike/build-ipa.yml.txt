name: Build Sugar Strike IPA

on:
  push:
    branches: [main, master]
  workflow_dispatch:

permissions:
  contents: write

jobs:
  build:
    runs-on: macos-latest
    steps:
      - name: Get the code
        uses: actions/checkout@v4

      - name: Show Xcode version
        run: xcodebuild -version

      - name: Build the app (unsigned)
        run: |
          xcodebuild \
            -project SugarStrike.xcodeproj \
            -target SugarStrike \
            -configuration Release \
            -sdk iphoneos \
            SYMROOT="$PWD/build" \
            CODE_SIGNING_ALLOWED=NO \
            CODE_SIGNING_REQUIRED=NO \
            CODE_SIGN_IDENTITY="" \
            build

      - name: Check the app icon made it in
        run: |
          APP="build/Release-iphoneos/SugarStrike.app"
          ls "$APP"
          test -f "$APP/Assets.car" || { echo "ERROR: icon catalog (Assets.car) missing"; exit 1; }
          ls "$APP"/AppIcon*.png >/dev/null 2>&1 || { echo "ERROR: AppIcon PNGs missing"; exit 1; }
          /usr/libexec/PlistBuddy -c "Print :CFBundleIcons" "$APP/Info.plist" || { echo "ERROR: icon not listed in Info.plist"; exit 1; }
          echo "App icon OK"

      - name: Package into SugarStrike.ipa
        run: |
          APP="build/Release-iphoneos/SugarStrike.app"
          test -d "$APP" || { echo "App not found"; ls -R build | head -50; exit 1; }
          mkdir -p Payload
          cp -R "$APP" Payload/
          zip -qry SugarStrike.ipa Payload
          ls -lh SugarStrike.ipa

      - name: Upload IPA (Actions tab download)
        uses: actions/upload-artifact@v4
        with:
          name: SugarStrike-ipa
          path: SugarStrike.ipa

      - name: Publish IPA to Releases (direct download link)
        env:
          GH_TOKEN: ${{ github.token }}
        run: |
          TAG="build-${{ github.run_number }}"
          gh release create "$TAG" SugarStrike.ipa \
            --title "Sugar Strike build ${{ github.run_number }}" \
            --notes "Unsigned IPA. Install it with Sideloadly or AltStore, which sign it with your Apple ID." \
            --latest
